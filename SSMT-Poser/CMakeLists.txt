cmake_minimum_required(VERSION 3.24)
project(SSMTPoser LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "SSMT-Poser requires Windows x64 and MSVC (SEH)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# 复用工作区已有 MinHook；独立编译进探针，不加载 Player Tweaks 插件。
if(NOT TARGET minhook)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../SSMT-Player-Tweaks/ThirdParty/MinHook"
        "${CMAKE_CURRENT_BINARY_DIR}/PoserMinHook")
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/Bullet/SSMT.cmake")
add_library(SSMT-Poser SHARED Plugin.cpp RuntimeProbe.cpp GenshinNativeRuntime.cpp LiveUnityProbe.cpp RigProbe.cpp BoneWriteProbe.cpp IKTimingProbe.cpp VmdMotion.cpp FacePlayback.cpp MotionPlayback.cpp NativePhysicsCapture.cpp SecondaryPhysics.cpp BulletPhysics.cpp PhysicsMotion.cpp UiBridge.cpp PoserUi.cpp CursorInput.cpp)
target_link_libraries(SSMT-Poser PRIVATE bcrypt user32 minhook PoserBulletBulletDynamics comctl32 dwmapi gdi32 shell32 ole32 uuid)
target_compile_options(SSMT-Poser PRIVATE /utf-8 /W4 /EHsc)
target_compile_definitions(SSMT-Poser PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set(SSMT_POSER_ARTIFACT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../dist/$<CONFIG>"
    CACHE STRING "探针 DLL 输出目录，可在实机占用旧产物时指定新目录")
add_custom_command(TARGET SSMT-Poser POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${SSMT_POSER_ARTIFACT_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SSMT-Poser>"
        "${SSMT_POSER_ARTIFACT_DIR}/SSMT-Poser.dll")

option(SSMT_POSER_BUILD_TESTS "Build local Poser test harnesses when present" ON)
if(SSMT_POSER_BUILD_TESTS)
    enable_testing()
    add_executable(PoserCursorTests Validation/CursorTests.cpp CursorInput.cpp)
    target_link_libraries(PoserCursorTests PRIVATE user32 minhook)
    target_compile_options(PoserCursorTests PRIVATE /utf-8 /W4 /EHsc)
    target_compile_definitions(PoserCursorTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(PoserCursorTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME PoserCursorTests COMMAND PoserCursorTests)
    add_executable(PoserPhysicsUiTests Validation/PhysicsUiTests.cpp UiBridge.cpp PoserUi.cpp CursorInput.cpp)
    target_link_libraries(PoserPhysicsUiTests PRIVATE user32 minhook comctl32 dwmapi gdi32 shell32 ole32 uuid gdiplus)
    target_compile_options(PoserPhysicsUiTests PRIVATE /utf-8 /W4 /EHsc)
    target_compile_definitions(PoserPhysicsUiTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(PoserPhysicsUiTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME PoserPhysicsUiTests COMMAND PoserPhysicsUiTests)
    add_executable(PoserPhysicsTests Validation/PhysicsTests.cpp Validation/FaceTests.cpp Validation/SkinSurfaceTests.cpp Validation/PhysicsMotionFixture.cpp SecondaryPhysics.cpp BulletPhysics.cpp PhysicsMotion.cpp RuntimeProbe.cpp GenshinNativeRuntime.cpp
        LiveUnityProbe.cpp RigProbe.cpp BoneWriteProbe.cpp IKTimingProbe.cpp VmdMotion.cpp FacePlayback.cpp MotionPlayback.cpp NativePhysicsCapture.cpp UiBridge.cpp)
    target_link_libraries(PoserPhysicsTests PRIVATE bcrypt user32 minhook PoserBulletBulletDynamics)
    target_compile_options(PoserPhysicsTests PRIVATE /utf-8 /W4 /EHsc)
    target_compile_definitions(PoserPhysicsTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX SSMT_POSER_TRACE_PHYSICS=1)
    target_include_directories(PoserPhysicsTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME PoserPhysicsTests COMMAND PoserPhysicsTests)
    # IK 语义回归属于源码；不依赖本机被忽略的调查用 Tests 目录。
    add_executable(PoserLegIKTests Validation/LegIKTests.cpp RuntimeProbe.cpp GenshinNativeRuntime.cpp
        LiveUnityProbe.cpp RigProbe.cpp BoneWriteProbe.cpp IKTimingProbe.cpp VmdMotion.cpp FacePlayback.cpp MotionPlayback.cpp NativePhysicsCapture.cpp SecondaryPhysics.cpp BulletPhysics.cpp PhysicsMotion.cpp UiBridge.cpp)
    target_link_libraries(PoserLegIKTests PRIVATE bcrypt user32 minhook PoserBulletBulletDynamics)
    target_compile_options(PoserLegIKTests PRIVATE /utf-8 /W4 /EHsc)
    target_compile_definitions(PoserLegIKTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(PoserLegIKTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME PoserLegIKTests COMMAND PoserLegIKTests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/ProbeTests.cpp" AND
       EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/BasisTests.cpp" AND
       EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/AvatarTests.cpp")
        add_executable(PoserProbeTests Tests/ProbeTests.cpp Tests/BasisTests.cpp Tests/AvatarTests.cpp RuntimeProbe.cpp GenshinNativeRuntime.cpp LiveUnityProbe.cpp RigProbe.cpp BoneWriteProbe.cpp IKTimingProbe.cpp VmdMotion.cpp FacePlayback.cpp MotionPlayback.cpp NativePhysicsCapture.cpp SecondaryPhysics.cpp BulletPhysics.cpp PhysicsMotion.cpp UiBridge.cpp)
        target_link_libraries(PoserProbeTests PRIVATE bcrypt user32 minhook PoserBulletBulletDynamics)
        target_compile_options(PoserProbeTests PRIVATE /utf-8 /W4 /EHsc)
        target_compile_definitions(PoserProbeTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_include_directories(PoserProbeTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME PoserProbeTests COMMAND PoserProbeTests "$<TARGET_FILE:SSMT-Poser>")
        add_test(NAME PoserBasisTests COMMAND PoserProbeTests --basis)
        add_test(NAME PoserAvatarTests COMMAND PoserProbeTests --avatar)
    else()
        message(STATUS "Poser probe test sources are absent; skipping PoserProbeTests")
    endif()

    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/UiTests.cpp")
        add_executable(PoserUiTests Tests/UiTests.cpp UiBridge.cpp PoserUi.cpp CursorInput.cpp)
        target_link_libraries(PoserUiTests PRIVATE user32 minhook comctl32 dwmapi gdi32 shell32 ole32 uuid gdiplus)
        target_compile_options(PoserUiTests PRIVATE /utf-8 /W4 /EHsc)
        target_compile_definitions(PoserUiTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_include_directories(PoserUiTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME PoserUiTests COMMAND PoserUiTests)
    else()
        message(STATUS "Poser UI test source is absent; skipping PoserUiTests")
    endif()
endif()
